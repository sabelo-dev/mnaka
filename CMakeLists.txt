cmake_minimum_required(VERSION 3.24)
project(mnaka VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Concurrent Network)
qt_standard_project_setup(REQUIRES 6.8)
set(QT_QML_GENERATE_QMLLS_INI ON)
include(CTest)
option(MNAKA_BUILD_RELAY "Build the internet relay (not yet implemented)" OFF)
if(MNAKA_BUILD_RELAY)
    message(FATAL_ERROR "Relay HTTP/authentication targets are not implemented or qualified yet")
endif()
add_subdirectory(core)
add_subdirectory(adapters)
add_subdirectory(engines)
if(NOT ANDROID AND NOT IOS AND NOT EMSCRIPTEN)
    find_package(Qt6 REQUIRED COMPONENTS Sql)
    qt_add_library(mnaka_storage_pg STATIC core/storage/PgConnection.h core/storage/PgConnection.cpp
        core/storage/ProjectRepository.h core/storage/ProjectRepository.cpp
        core/storage/MigrationRunner.h core/storage/MigrationRunner.cpp)
    target_include_directories(mnaka_storage_pg PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(mnaka_storage_pg PUBLIC Qt6::Sql)
    if(WIN32)
        target_link_libraries(mnaka_storage_pg PRIVATE Advapi32)
    endif()
    qt_add_executable(mnaka_migrate apps/migrate/main.cpp)
    target_link_libraries(mnaka_migrate PRIVATE mnaka_storage_pg)
    install(TARGETS mnaka_migrate RUNTIME DESTINATION bin)
endif()
add_subdirectory(apps/client)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(workspace_tests tests/workspace_tests.cpp)
    target_link_libraries(workspace_tests PRIVATE Qt6::Test mnaka_client_module)
    qt_add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE Qt6::Test mnaka_core)
    add_test(NAME broker_security_contract COMMAND core_tests)
    qt_add_executable(orchestrator_tests tests/orchestrator_tests.cpp)
    target_link_libraries(orchestrator_tests PRIVATE Qt6::Test mnaka_engines)
    add_test(NAME task_orchestrator_contract COMMAND orchestrator_tests)
    qt_add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE Qt6::Test mnaka_engines)
    add_test(NAME async_engine_contract COMMAND engine_tests)
    qt_add_executable(skill_tests tests/skill_tests.cpp)
    target_link_libraries(skill_tests PRIVATE Qt6::Test mnaka_engines)
    add_test(NAME declarative_skill_contract COMMAND skill_tests)
    qt_add_executable(auth_tests tests/auth_tests.cpp)
    target_link_libraries(auth_tests PRIVATE Qt6::Test Qt6::Network mnaka_adapters)
    add_test(NAME supabase_auth_contract COMMAND auth_tests)
    if(TARGET mnaka_storage_pg)
        qt_add_executable(storage_tests tests/storage_tests.cpp)
        target_link_libraries(storage_tests PRIVATE Qt6::Test mnaka_storage_pg)
        add_test(NAME postgres_fail_closed COMMAND storage_tests)
    endif()
    # The Supabase CLI applies its own copies of the relay migrations; they must not drift.
    add_test(NAME supabase_relay_bootstrap_in_sync COMMAND ${CMAKE_COMMAND} -E compare_files
        ${CMAKE_CURRENT_SOURCE_DIR}/migrations/relay-postgres/001_bootstrap.sql
        ${CMAKE_CURRENT_SOURCE_DIR}/supabase/migrations/20261010000001_relay_bootstrap.sql)
    add_test(NAME supabase_relay_policies_in_sync COMMAND ${CMAKE_COMMAND} -E compare_files
        ${CMAKE_CURRENT_SOURCE_DIR}/migrations/relay-postgres/002_owner_policies.sql
        ${CMAKE_CURRENT_SOURCE_DIR}/supabase/migrations/20261010000002_relay_owner_policies.sql)
    add_test(NAME workspace_contract COMMAND workspace_tests)
    add_test(NAME workspace_qml_smoke COMMAND mnaka_client --smoke-test)
    add_test(NAME workspace_qml_mock_task COMMAND mnaka_client --smoke-test --mock-session)
    set_tests_properties(workspace_qml_smoke workspace_qml_mock_task PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 15 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
